# SEC-07 KEV·EPSS and scan-time display Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Attach CISA KEV and FIRST EPSS evidence to each running-image vulnerability, show it as the default priority order in the console, and make the "수집 / 검사 시각" cell read clearly.

**Architecture:** The platform API (Go) reads KEV (whole catalog, daily, background refresh) and EPSS (per CVE, 24h cache) when `GET /v1/security/vulnerabilities` builds its report, waits at most 8s, and adds `exploitation` per vulnerability and `exploitSources` per report. The console (React) maps them to chips, a priority sort and reasons, and lists KEV/EPSS as rows of the sources tab.

**Tech Stack:** Go + oapi-codegen (`go generate ./pkg/apiv1`), React + TypeScript + vitest, openapi-typescript 6 (`npm run gen:api`).

**Spec:** `docs/superpowers/specs/2026-11-07-sec07-kev-epss-design.md`

## Global Constraints

- KEV feed: `https://www.cisa.gov/sites/default/files/feeds/known_exploited_vulnerabilities.json`; refresh every 24h; last good catalog used up to 3 days (72h), then `unknown`.
- EPSS: `https://api.first.org/data/v1/epss?cve=A,B`; at most 100 CVEs per request; 24h cache; a cached score older than 72h is `unknown`. EPSS numbers arrive as strings.
- A report waits at most 8s for KEV/EPSS. No failure fails the vulnerability list.
- CVE: the advisory if it starts with `CVE-`, else the first `CVE-` alias, else none (`no_cve`, "CVE 없음").
- Never change `severity`, never combine scores, no VEX field (ADR states "VEX 미적용").
- Never print credentials. Korean UI copy exactly as written below.
- Platform worktree: `.worktrees/platform-security-sec05`, branch `codex/sec07-kev-epss`. Frontend worktree: `.worktrees/frontend-security-sec06`, new branch `codex/sec07-exploit-ui` from `origin/main`.
- Platform tests: `go test -race ./internal/security/ ./internal/api/` (Postgres-backed packages run in CI). Frontend: `npx vitest run`, `npx tsc -b --noEmit` (Node at `/c/Program Files/nodejs`).

package security

import (

// SEC-07 (ADR 0034): exploitation evidence for image vulnerabilities, read
// when the vulnerability report is built. CISA KEV says a CVE is known to be
// exploited; FIRST EPSS estimates the probability of exploitation in the next
// 30 days. Neither changes a severity, and they are never combined.

const (
	SourceKEV  = "cisa-kev"
	SourceEPSS = "first-epss"
	// ExploitWait bounds how long a report waits for KEV and EPSS.
	ExploitWait = 8 * time.Second

	kevFeed         = "https://www.cisa.gov/sites/default/files/feeds/known_exploited_vulnerabilities.json"
	epssAPI         = "https://api.first.org/data/v1/epss"
	kevRefresh      = 24 * time.Hour
	exploitMaxAge   = 72 * time.Hour
	exploitRetry    = 15 * time.Minute
	kevFetchTimeout = 60 * time.Second
	epssTTL         = 24 * time.Hour
	epssBatch       = 100
)

type kevEntry struct {
	dateAdded, dueDate string
	ransomware         bool
}

type epssEntry struct {
	scored                  bool
	probability, percentile float64
	date                    string
	at                      time.Time
}

// ExploitIntel caches the KEV catalog and EPSS scores. It is safe for
// concurrent use.
type ExploitIntel struct {
	kevURL, epssURL string
	client          *http.Client
	now             func() time.Time

	mu         sync.Mutex
	kev        map[string]kevEntry
	kevVersion string
	kevAt      time.Time // last successful read
	kevTried   time.Time
	kevErr     string
	kevLoading chan struct{}
	epss       map[string]epssEntry
	epssAt     time.Time
	epssErr    string
	epssDate   string
}

// NewExploitIntel reads the public KEV feed and EPSS API.
func NewExploitIntel() *ExploitIntel {
	return &ExploitIntel{kevURL: kevFeed, epssURL: epssAPI, client: &http.Client{}, now: time.Now, epss: map[string]epssEntry{}}
}

// AttachExploitation sets exploitation on each vulnerability of r and adds
// exploitSources. It waits at most ExploitWait; what is not read in time is
// unknown. A nil x leaves r unchanged.
func AttachExploitation(ctx context.Context, r *apiv1.SecurityVulnerabilityReport, x *ExploitIntel) {
	if x == nil {
		return
	}
	ctx, cancel := context.WithTimeout(ctx, ExploitWait)
	defer cancel()
	set := map[string]bool{}
	for _, v := range r.Vulnerabilities {
		if c := vulnCVE(v); c != "" {
			set[c] = true
		}
	}
	var wg sync.WaitGroup
	wg.Add(2)
	go func() { defer wg.Done(); x.refreshKEV(ctx) }()
	go func() { defer wg.Done(); x.refreshEPSS(ctx, slices.Sorted(maps.Keys(set))) }()
	wg.Wait()

	x.mu.Lock()
	defer x.mu.Unlock()
	for i := range r.Vulnerabilities {
		e := x.exploitation(vulnCVE(r.Vulnerabilities[i]))
		r.Vulnerabilities[i].Exploitation = &e
	}
	src := x.sources(len(set))
	r.ExploitSources = &src
}

func vulnCVE(v apiv1.SecurityVulnerability) string {
	for _, id := range append([]string{v.Advisory}, v.Aliases...) {
		if len(id) > 4 && strings.EqualFold(id[:4], "CVE-") {
			return strings.ToUpper(id)
		}
	}
	return ""
}

// refreshKEV starts a catalog read when one is due, in the background so a
// slow feed still fills the cache for the next report, and waits for it until
// ctx ends.
func (x *ExploitIntel) refreshKEV(ctx context.Context) {
	x.mu.Lock()
	now := x.now()
	if x.kevLoading == nil && now.Sub(x.kevAt) >= kevRefresh && now.Sub(x.kevTried) >= exploitRetry {
		x.kevTried = now
		done := make(chan struct{})
		x.kevLoading = done
		go func() {
			defer close(done)
			fctx, cancel := context.WithTimeout(context.Background(), kevFetchTimeout)
			defer cancel()
			cat, version, err := x.readKEV(fctx)
			x.mu.Lock()
			defer x.mu.Unlock()
			x.kevLoading = nil
			if err != nil {
				x.kevErr = shorten(err.Error(), 160)
				return
			}
			x.kev, x.kevVersion, x.kevAt, x.kevErr = cat, version, x.now(), ""
		}()
	}
	wait := x.kevLoading
	x.mu.Unlock()
	if wait != nil {
		select {
		case <-wait:
		case <-ctx.Done():
		}
	}
}

func (x *ExploitIntel) readKEV(ctx context.Context) (map[string]kevEntry, string, error) {
	var body struct {
		CatalogVersion  string `json:"catalogVersion"`
		Vulnerabilities []struct {
			CVE        string `json:"cveID"`
			DateAdded  string `json:"dateAdded"`
			DueDate    string `json:"dueDate"`
			Ransomware string `json:"knownRansomwareCampaignUse"`
		} `json:"vulnerabilities"`
	}
	if err := x.getJSON(ctx, x.kevURL, 32<<20, &body); err != nil {
		return nil, "", fmt.Errorf("KEV catalog: %w", err)
	}
	if body.CatalogVersion == "" || len(body.Vulnerabilities) == 0 {
		return nil, "", errors.New("KEV catalog: empty or unrecognized")
	}
	out := make(map[string]kevEntry, len(body.Vulnerabilities))
	for _, v := range body.Vulnerabilities {
		out[strings.ToUpper(v.CVE)] = kevEntry{dateAdded: v.DateAdded, dueDate: v.DueDate, ransomware: v.Ransomware == "Known"}
	}
	return out, body.CatalogVersion, nil
}

// refreshEPSS reads the CVEs with no score younger than epssTTL, 100 per
// request. On a failure the rest keep what they had.
func (x *ExploitIntel) refreshEPSS(ctx context.Context, cves []string) {
	x.mu.Lock()
	now := x.now()
	var need []string
	for _, c := range cves {
		if e, ok := x.epss[c]; !ok || now.Sub(e.at) >= epssTTL {
			need = append(need, c)
		}
	}
	x.mu.Unlock()
	for start := 0; start < len(need); start += epssBatch {
		batch := need[start:min(start+epssBatch, len(need))]
		got, err := x.readEPSS(ctx, batch)
		x.mu.Lock()
		if err != nil {
			x.epssErr = shorten(err.Error(), 160)
			x.mu.Unlock()
			return
		}
		at := x.now()
		for _, c := range batch {
			e := got[c] // absent: FIRST has no score
			e.at = at
			x.epss[c] = e
			if e.date != "" {
				x.epssDate = e.date
			}
		}
		x.epssAt, x.epssErr = at, ""
		x.mu.Unlock()
	}
}

func (x *ExploitIntel) readEPSS(ctx context.Context, cves []string) (map[string]epssEntry, error) {
	var body struct {
		Data []struct {
			CVE        string `json:"cve"`
			EPSS       string `json:"epss"`
			Percentile string `json:"percentile"`
			Date       string `json:"date"`
		} `json:"data"`
	}
	q := url.Values{"cve": {strings.Join(cves, ",")}}
	if err := x.getJSON(ctx, x.epssURL+"?"+q.Encode(), 4<<20, &body); err != nil {
		return nil, fmt.Errorf("EPSS: %w", err)
	}
	out := map[string]epssEntry{}
	for _, d := range body.Data {
		p, err1 := strconv.ParseFloat(d.EPSS, 64)
		pct, err2 := strconv.ParseFloat(d.Percentile, 64)
		if err1 != nil || err2 != nil {
			return nil, fmt.Errorf("EPSS: unreadable score for %s", d.CVE)
		}
		out[strings.ToUpper(d.CVE)] = epssEntry{scored: true, probability: p, percentile: pct, date: d.Date}
	}
	return out, nil
}

func (x *ExploitIntel) exploitation(cve string) apiv1.SecurityExploitation {
	if cve == "" {
		return apiv1.SecurityExploitation{Kev: apiv1.SecurityKEV{State: apiv1.SecurityKEVStateNoCve}, Epss: apiv1.SecurityEPSS{State: apiv1.SecurityEPSSStateNoCve}}
	}
	now := x.now()
	out := apiv1.SecurityExploitation{Cve: &cve}
	if x.kev == nil || now.Sub(x.kevAt) > exploitMaxAge {
		out.Kev.State = apiv1.SecurityKEVStateUnknown
	} else {
		v := x.kevVersion
		out.Kev.CatalogVersion = &v
		out.Kev.State = apiv1.SecurityKEVStateNotListed
		if e, ok := x.kev[cve]; ok {
			added, due, ransom := e.dateAdded, e.dueDate, e.ransomware
			out.Kev.State, out.Kev.DateAdded, out.Kev.DueDate, out.Kev.Ransomware = apiv1.SecurityKEVStateListed, &added, &due, &ransom
		}
	}
	switch e, ok := x.epss[cve]; {
	case !ok || now.Sub(e.at) > exploitMaxAge:
		out.Epss.State = apiv1.SecurityEPSSStateUnknown
	case !e.scored:
		out.Epss.State = apiv1.SecurityEPSSStateNotScored
	default:
		p, pct, d := e.probability, e.percentile, e.date
		out.Epss.State, out.Epss.Probability, out.Epss.Percentile, out.Epss.Date = apiv1.SecurityEPSSStateScored, &p, &pct, &d
	}
	return out
}

	case x.epssAt.IsZero() || now.Sub(x.epssAt) > exploitMaxAge:

// SEC-07: default priority order and its reasons. KEV, running state and EPSS
// are separate bases next to severity; nothing here changes a severity.
import type { SecuritySeverity, VulnerabilityFinding } from "./security";

const SEVERITY_RANK: Record<SecuritySeverity, number> = { critical: 0, high: 1, medium: 2, low: 3, info: 4, unknown: 5 };
const md = (d: string) => d.slice(5).replace("-", "/");
const listed = (g: VulnerabilityFinding[]) => g.some((v) => v.kev === true);
const isRunning = (g: VulnerabilityFinding[]) => g.some((v) => v.occurrences.some((o) => o.running === true));
const maxEpss = (g: VulnerabilityFinding[]) => Math.max(-1, ...g.map((v) => v.epss ?? -1));
const topPercent = (pct: number) => Math.max(1, Math.round((1 - pct) * 100));

/** KEV listed → running → severity → EPSS (highest first) → advisory. KEV unknown ranks as not listed. */
export function sortByPriority(groups: VulnerabilityFinding[][]): VulnerabilityFinding[][] {
  const key = (g: VulnerabilityFinding[]) => [listed(g) ? 0 : 1, isRunning(g) ? 0 : 1, Math.min(...g.map((v) => SEVERITY_RANK[v.severity]))];
  return [...groups].sort((a, b) => {
    const ka = key(a), kb = key(b);
    for (let i = 0; i < ka.length; i++) if (ka[i] !== kb[i]) return ka[i] - kb[i];
    return maxEpss(b) - maxEpss(a) || a[0].advisory.localeCompare(b[0].advisory);
  });
}

export function priorityReasons(group: VulnerabilityFinding[]): string[] {
  const out: string[] = [];
  if (listed(group)) out.push("KEV 등재");
  if (group.some((v) => v.ransomware)) out.push("랜섬웨어 사용");
  if (isRunning(group)) out.push("실행 중");
  const pct = Math.max(-1, ...group.map((v) => v.epssPercentile ?? -1));
  if (pct >= 0.9) out.push(`EPSS 상위 ${topPercent(pct)}%`);
  if (group.some((v) => v.fixedVersion)) out.push("수정 버전 있음");
  return out;
}

export function kevLabel(v: VulnerabilityFinding): string {
  if (v.kevState === "no_cve") return "CVE 없음";
  if (v.kev === true) return v.kevDueDate ? `KEV 등재 (기한 ${md(v.kevDueDate)})` : "KEV 등재";
  return v.kev === false ? "KEV 미등재" : "KEV 미확인";
}

export function epssLabel(v: VulnerabilityFinding, demo = false): string {
  if (v.epssState === "no_cve") return "CVE 없음";
  if (v.epssState === "not_scored") return "EPSS 점수 없음";
  if (v.epss === null) return "EPSS 미확인";
  const top = v.epssPercentile != null ? ` (상위 ${topPercent(v.epssPercentile)}%)` : "";
  const day = v.epssDate ? ` · ${md(v.epssDate)} 기준` : "";
  return `EPSS ${(v.epss * 100).toFixed(1)}%${top}${day}${demo ? " (예시)" : ""}`;
}
